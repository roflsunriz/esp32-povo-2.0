cmake_minimum_required(VERSION 3.20)
project(povo_cyd_tests LANGUAGES CXX)
enable_testing()
set(ARDUINOJSON_DIR "${CMAKE_SOURCE_DIR}/.pio/libdeps/cyd/ArduinoJson/src" CACHE PATH "ArduinoJson include directory")
add_executable(status-test test/status-test.cpp)
target_include_directories(status-test PRIVATE include "${ARDUINOJSON_DIR}")
target_compile_features(status-test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(status-test PRIVATE /W4 /WX /utf-8 /EHsc)
else()
  target_compile_options(status-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME status COMMAND status-test)
add_executable(ui-test test/ui-test.cpp)
target_include_directories(ui-test PRIVATE test/ui-stubs include)
target_compile_features(ui-test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(ui-test PRIVATE /W4 /WX /utf-8 /EHsc)
else()
  target_compile_options(ui-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME ui COMMAND ui-test)
add_executable(display-settings-test test/display-settings-test.cpp)
target_include_directories(display-settings-test PRIVATE include)
target_compile_features(display-settings-test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(display-settings-test PRIVATE /W4 /WX /utf-8 /EHsc)
else()
  target_compile_options(display-settings-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME display-settings COMMAND display-settings-test)
foreach(test_name auth-protocol direct-status)
  add_executable(${test_name}-test test/${test_name}-test.cpp)
  target_include_directories(${test_name}-test PRIVATE include "${ARDUINOJSON_DIR}")
  target_compile_features(${test_name}-test PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(${test_name}-test PRIVATE /W4 /WX /utf-8 /EHsc)
  else()
    target_compile_options(${test_name}-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME ${test_name} COMMAND ${test_name}-test)
endforeach()
add_executable(client-test test/client-test.cpp)
target_include_directories(client-test PRIVATE test/client-stubs include "${ARDUINOJSON_DIR}")
